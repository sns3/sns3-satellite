91Ptr<SatOutputFileStreamDoubleContainer>
92SatFadingOutputTraceContainer::AddNode(key_t key)
93{
94 NS_LOG_FUNCTION(this);
95
96 std::stringstream filename;
97 std::string dataPath = SatEnvVariables::GetInstance()->GetOutputPath();
98
99 int32_t gwId = Singleton<SatIdMapper>::Get()->GetGwIdWithMac(key.first);
100 int32_t utId = Singleton<SatIdMapper>::Get()->GetUtIdWithMac(key.first);
101 int32_t beamId = Singleton<SatIdMapper>::Get()->GetBeamIdWithMac(key.first);
102
103 if (beamId < 0 || (utId < 0 && gwId < 0))
104 {
105 return nullptr;
106 }
107 else
108 {
109 if (utId >= 0 && gwId < 0)
110 {
111 filename << dataPath << "/fading_output_trace_BEAM_" << beamId << "_UT_" << utId
112 << "_channelType_" << SatEnums::GetChannelTypeName(key.second);
113 }
114
115 if (gwId >= 0 && utId < 0)
116 {
117 filename << dataPath << "/fading_output_trace_BEAM_" << beamId << "_GW_" << gwId
118 << "_channelType_" << SatEnums::GetChannelTypeName(key.second);
119 }
120
121 std::pair<container_t::iterator, bool> result = m_container.insert(
122 std::make_pair(key,
123 CreateObject<SatOutputFileStreamDoubleContainer>(
124 filename.str().c_str(),
125 std::ios::out,
126 SatBaseTraceContainer::FADING_TRACE_DEFAULT_NUMBER_OF_COLUMNS)));
127
128 if (result.second == false)
129 {
130 NS_FATAL_ERROR("SatFadingOutputTraceContainer::AddNode failed");
131 }
132
133 NS_LOG_INFO("Added node with MAC " << key.first << " channel type " << key.second);
134
135 return result.first->second;
136 }
137}

139Ptr<SatOutputFileStreamDoubleContainer>
140SatFadingOutputTraceContainer::FindNode(key_t key)
141{
142 NS_LOG_FUNCTION(this);
143
144 container_t::iterator iter = m_container.find(key);
145
146 if (iter == m_container.end())
147 {
148 return AddNode(key);
149 }
150
151 return iter->second;
152}

154void
155SatFadingOutputTraceContainer::WriteToFile()
156{
157 NS_LOG_FUNCTION(this);
158
159 container_t::iterator iter;
160
161 for (iter = m_container.begin(); iter != m_container.end(); iter++)
162 {
163 if (m_enableFigureOutput)
164 {
165 iter->second->EnableFigureOutput("Fading trace",
166 "Time (s)",
167 "Fading (dB)",
168 "set key top right",
169 SatOutputFileStreamDoubleContainer::DECIBEL_AMPLITUDE,
170 Gnuplot2dDataset::LINES);
171 }
172 iter->second->WriteContainerToFile();
173 }
174}

176void
177SatFadingOutputTraceContainer::AddToContainer(key_t key, std::vector<double> newItem)
178{
179 NS_LOG_FUNCTION(this);
180
181 if (newItem.size() != SatBaseTraceContainer::FADING_TRACE_DEFAULT_NUMBER_OF_COLUMNS)
182 {
183 NS_FATAL_ERROR("SatFadingOutputTraceContainer::AddToContainer - Incorrect vector size");
184 }
185
186 Ptr<SatOutputFileStreamDoubleContainer> node = FindNode(key);
187
188 if (node != nullptr)
189 {
190 node->AddToContainer(newItem);
191 }
192}
