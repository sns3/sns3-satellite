147 utApps.Start(Seconds(1.0));
148 utApps.Stop(Seconds(2.1));

152 Address(InetSocketAddress(helper->GetUserAddress(gwUsers.Get(0)), port)));

155 gwApps.Start(Seconds(1.0));
156 gwApps.Stop(Seconds(3.0));
157
158 Simulator::Stop(Seconds(11));
159 Simulator::Run();
160
161 Simulator::Destroy();
162
163 Ptr<PacketSink> receiver = DynamicCast<PacketSink>(gwApps.Get(0));
164 Ptr<CbrApplication> sender = DynamicCast<CbrApplication>(utApps.Get(0));
165
166 // here we check that results are as expected.
167 // * Sender has sent something
168 // * Receiver got all all data sent
169 NS_TEST_ASSERT_MSG_NE(sender->GetSent(), (uint32_t)0, "Nothing sent !");
170 NS_TEST_ASSERT_MSG_EQ(receiver->GetTotalRx(), sender->GetSent(), "Packets were lost !");

172 SatEnvVariables::GetInstance()->DoDispose();

174 // <<< End of actual test using Simple scenario <<<
