242SatMac::SetTimeTag(SatPhy::PacketContainer_t packets)
243{
244 if (m_isStatisticsTagsEnabled)
245 {
246 for (SatPhy::PacketContainer_t::const_iterator it = packets.begin(); it != packets.end();
247 ++it)
248 {
249 SatMacTimeTag timeTag;
250 if (!(*it)->PeekPacketTag(timeTag))
251 {
252 (*it)->AddPacketTag(SatMacTimeTag(Simulator::Now()));
253 }
254
255 SatMacLinkTimeTag linkTimeTag;
256 if (!(*it)->PeekPacketTag(linkTimeTag))
257 {
258 (*it)->AddPacketTag(SatMacLinkTimeTag(Simulator::Now()));

264void
265SatMac::SendPacket(SatPhy::PacketContainer_t packets,
266 uint32_t carrierId,
267 Time duration,
268 SatSignalParameters::txInfo_s txInfo)
269{
270 NS_LOG_FUNCTION(this);
271
272 // Add a SatMacTimeTag tag for packet delay computation at the receiver end.
273 SetTimeTag(packets);
274
275 // Update local destination MAC tag with satellite one if satellite is regenerative
276 if (m_isRegenerative)
277 {
278 for (SatPhy::PacketContainer_t::const_iterator it = packets.begin(); it != packets.end();
279 ++it)
280 {
281 SatMacTag mTag;
282 bool success = (*it)->RemovePacketTag(mTag);
283
284 // MAC tag found
285 if (success)
286 {
287 mTag.SetDestAddress(Mac48Address::ConvertFrom(m_satelliteAddress));
288 (*it)->AddPacketTag(mTag);
289 }
290 }
291 }
292
299 for (SatPhy::PacketContainer_t::const_iterator it = packets.begin(); it != packets.end(); ++it)
300 {
301 SatControlMsgTag cTag;
302 bool success = (*it)->RemovePacketTag(cTag);
303
304 // Control tag found
305 if (success)
306 {
307 uint32_t sendId = cTag.GetMsgId();
308
309 // Add the msg to container and receive the used recv id.
310 uint32_t recvId = SendCtrlMsgFromContainer(sendId);
311
312 // Store the recv id to tag and add the tag back to the packet
313 cTag.SetMsgId(recvId);
314 (*it)->AddPacketTag(cTag);
315
316 if (cTag.GetMsgType() == SatControlMsgTag::SAT_NCR_CTRL_MSG)
317 {
318 Ptr<SatNcrMessage> ncrMsg = m_ncrMessagesToSend.front();
319 m_ncrMessagesToSend.pop();
320 uint8_t lastSOFSize = m_ncrV2 ? 3 : 1;
321 ncrMsg->SetNcrDate(m_lastSOF.size() == lastSOFSize
322 ? m_lastSOF.front().GetNanoSeconds() * 0.027
323 : 0);
324 }
325 }
326 }
327
328 // Use call back to send packet to lower layer
329 m_txCallback(packets, carrierId, duration, txInfo);
330}

332void
333SatMac::RxTraces(SatPhy::PacketContainer_t packets)
334{
335 NS_LOG_FUNCTION(this);
336
337 if (m_isStatisticsTagsEnabled)
338 {
339 for (SatPhy::PacketContainer_t::const_iterator it1 = packets.begin(); it1 != packets.end();
340 ++it1)
341 {
342 // Remove packet tag
343 SatMacTag macTag;
344 bool mSuccess = (*it1)->PeekPacketTag(macTag);
345 if (!mSuccess)
346 {
347 NS_FATAL_ERROR("MAC tag was not found from the packet!");
348 }
349
350 // If the packet is intended for this receiver
351 Mac48Address destAddress = macTag.GetDestAddress();
352
353 if (destAddress == m_nodeInfo->GetMacAddress())
354 {
355 Address addr; // invalid address.
356
357 bool isTaggedWithAddress = false;
358 ByteTagIterator it2 = (*it1)->GetByteTagIterator();
359
360 while (!isTaggedWithAddress && it2.HasNext())
361 {
362 ByteTagIterator::Item item = it2.Next();
363
364 if (item.GetTypeId() == SatAddressTag::GetTypeId())
365 {
366 NS_LOG_DEBUG(this << " contains a SatAddressTag tag:" << " start="
367 << item.GetStart() << " end=" << item.GetEnd());
368 SatAddressTag addrTag;
369 item.GetTag(addrTag);
370 addr = addrTag.GetSourceAddress();
371 isTaggedWithAddress = true; // this will exit the while loop.
372 }
373 }
374
375 m_rxTrace(*it1, addr);
376
377 SatMacTimeTag timeTag;
378 if ((*it1)->RemovePacketTag(timeTag))
379 {
380 NS_LOG_DEBUG(this << " contains a SatMacTimeTag tag");
381 Time delay = Simulator::Now() - timeTag.GetSenderTimestamp();
382 m_rxDelayTrace(delay, addr);
383 if (m_lastDelay.IsZero() == false)
384 {
385 Time jitter = Abs(delay - m_lastDelay);
386 m_rxJitterTrace(jitter, addr);
387 }
388 m_lastDelay = delay;
389 }
390 SatMacLinkTimeTag linkTimeTag;
391 if ((*it1)->RemovePacketTag(linkTimeTag))
392 {
393 NS_LOG_DEBUG(this << " contains a SatMacLinkTimeTag tag");
394 Time delay = Simulator::Now() - linkTimeTag.GetSenderLinkTimestamp();
395 m_rxLinkDelayTrace(delay, addr);
396 if (m_lastLinkDelay.IsZero() == false)
397 {
398 Time jitter = Abs(delay - m_lastLinkDelay);
399 m_rxLinkJitterTrace(jitter, addr);
400 }
401 m_lastLinkDelay = delay;
402 }
403 } // end of `if (destAddress == m_nodeInfo->GetMacAddress () || destAddress.IsBroadcast
404 // ())`
405 } // end of `for it1 = packets.begin () -> packets.end ()`
406 } // end of `if (m_isStatisticsTagsEnabled)`
407}
