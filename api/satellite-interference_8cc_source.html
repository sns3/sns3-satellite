130std::vector<std::pair<double, double>>
131SatInterference::Calculate(Ptr<SatInterference::InterferenceChangeEvent> event)
132{
133 NS_LOG_FUNCTION(this);
134
135 if (m_currentlyReceiving > 1)
136 {
137 std::map<Ptr<SatInterference::InterferenceChangeEvent>, bool>::iterator iter;
138 bool wasCollisionReported = true;
139
140 for (iter = m_packetCollisions.begin(); iter != m_packetCollisions.end(); iter++)
141 {
142 if (!iter->second)
143 {
144 wasCollisionReported = false;
145 }
146 iter->second = true;
147 }
148 if (!wasCollisionReported)
149 {
150 NS_LOG_INFO("Packet collision!");
151 }
152 }
153
154 return DoCalculate(event);
155}

168void
169SatInterference::NotifyRxStart(Ptr<SatInterference::InterferenceChangeEvent> event)
170{
171 NS_LOG_FUNCTION(this);
172
173 m_currentlyReceiving++;
174
175 std::pair<std::map<Ptr<SatInterference::InterferenceChangeEvent>, bool>::iterator, bool> result;
176 result = m_packetCollisions.insert(std::make_pair(event, false));
177
178 if (!result.second)
179 {
180 NS_FATAL_ERROR("SatInterference::NotifyRxStart - Event already exists");
181 }
182
183 DoNotifyRxStart(event);
184}

186void
187SatInterference::NotifyRxEnd(Ptr<SatInterference::InterferenceChangeEvent> event)
188{
189 NS_LOG_FUNCTION(this);
190
191 if (m_currentlyReceiving > 0)
192 {
193 m_currentlyReceiving--;
194 }
195
196 m_packetCollisions.erase(event);
197
198 DoNotifyRxEnd(event);
199}

201bool
202SatInterference::HasCollision(Ptr<SatInterference::InterferenceChangeEvent> event)
203{
204 NS_LOG_FUNCTION(this);
205
206 std::map<Ptr<SatInterference::InterferenceChangeEvent>, bool>::iterator result =
207 m_packetCollisions.find(event);
208
209 if (result == m_packetCollisions.end())
210 {
211 NS_FATAL_ERROR("SatInterference::HasCollision - Event not found");
212 }
213
214 return result->second;
215}
