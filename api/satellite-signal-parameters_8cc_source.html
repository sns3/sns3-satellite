52SatSignalParameters::SatSignalParameters(const SatSignalParameters& p)
53{
54 for (PacketsInBurst_t::const_iterator i = p.m_packetsInBurst.begin();
55 i != p.m_packetsInBurst.end();
56 i++)
57 {
58 m_packetsInBurst.push_back((*i)->Copy());
59 }
60
61 m_satId = p.m_satId;
62 m_beamId = p.m_beamId;
63 m_carrierId = p.m_carrierId;
64 m_duration = p.m_duration;
65 m_phyTx = p.m_phyTx;
66 m_txPower_W = p.m_txPower_W;

72 m_txInfo.fecBlockSizeInBytes = p.m_txInfo.fecBlockSizeInBytes;
73 m_txInfo.frameType = p.m_txInfo.frameType;
