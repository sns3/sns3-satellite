370Gnuplot2dDataset
371SatLinkResultsPlot::GetGnuplotDataset(Ptr<SatLookUpTable> table, std::string title)
372{
373 Gnuplot2dDataset ret;
374 ret.SetTitle(title);
375 ret.SetStyle(Gnuplot2dDataset::LINES);
376
377 bool writeFlag = false;
378 double bler;
379
380 for (double sinr = m_minSinrDb; sinr < m_maxSinrDb; sinr += m_resolutionDb)
381 {
382 bler = table->GetBler(sinr);
383
384 if (!writeFlag)
385 {
386 if (bler < 1)
387 {
388 // first time error rate is not 100%
389 writeFlag = true; // start writing the plot
390 ret.Add(sinr - m_resolutionDb, 1.0); // also add the previous point
391 ret.Add(sinr, bler);
392 }
393 }
394 else
395 {
396 if (bler < 1e-10)
397 {
398 // close enough to 0% error rate
399 writeFlag = false; // stop writing the plot
400 sinr = m_maxSinrDb; // exit for
401 }
402 else
403 {
404 ret.Add(sinr, bler);
405 }
406 }
407
408 } // end of for (sinr = m_minSinrDb; sinr < m_maxSinrDb; sinr += m_resolutionDb)
409
410 return ret;
411
412} // end of GetGnuplotDataset
