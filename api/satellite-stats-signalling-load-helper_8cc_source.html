78{
79 NS_LOG_FUNCTION(this);
80
81 switch (GetOutputType())
82 {
83 case SatStatsHelper::OUTPUT_NONE:
84 NS_FATAL_ERROR(GetOutputTypeName(GetOutputType())
85 << " is not a valid output type for this statistics.");
86 break;
87
88 case SatStatsHelper::OUTPUT_SCALAR_FILE: {
89 // Setup aggregator.
90 m_aggregator = CreateAggregator("ns3::MultiFileAggregator",
91 "OutputFileName",
92 StringValue(GetOutputFileName()),
93 "MultiFileMode",
94 BooleanValue(false),
95 "EnableContextPrinting",
96 BooleanValue(true),
97 "GeneralHeading",
98 StringValue(GetIdentifierHeading("signalling_kbps")));
99
100 // Setup second-level collectors.
101 m_terminalCollectors.SetType("ns3::ScalarCollector");
102 m_terminalCollectors.SetAttribute("InputDataType",
103 EnumValue(ScalarCollector::INPUT_DATA_TYPE_DOUBLE));
104 m_terminalCollectors.SetAttribute(
105 "OutputType",
106 EnumValue(ScalarCollector::OUTPUT_TYPE_AVERAGE_PER_SECOND));
107 CreateCollectorPerIdentifier(m_terminalCollectors);
108 m_terminalCollectors.ConnectToAggregator("Output",
109 m_aggregator,
110 &MultiFileAggregator::Write1d);
111
112 // Setup first-level collectors.
113 m_conversionCollectors.SetType("ns3::UnitConversionCollector");
114 m_conversionCollectors.SetAttribute("ConversionType",

116 CreateCollectorPerIdentifier(m_conversionCollectors);
117 m_conversionCollectors.ConnectToCollector("Output",
118 m_terminalCollectors,
119 &ScalarCollector::TraceSinkDouble);
120 break;
121 }
122
123 case SatStatsHelper::OUTPUT_SCATTER_FILE: {
124 // Setup aggregator.
125 m_aggregator = CreateAggregator("ns3::MultiFileAggregator",
126 "OutputFileName",
127 StringValue(GetOutputFileName()),
128 "GeneralHeading",
129 StringValue(GetTimeHeading("signalling_kbps")));
130
131 // Setup second-level collectors.
132 m_terminalCollectors.SetType("ns3::IntervalRateCollector");
133 m_terminalCollectors.SetAttribute("InputDataType",
134 EnumValue(IntervalRateCollector::INPUT_DATA_TYPE_DOUBLE));
135 CreateCollectorPerIdentifier(m_terminalCollectors);
136 m_terminalCollectors.ConnectToAggregator("OutputWithTime",
137 m_aggregator,
138 &MultiFileAggregator::Write2d);
139 m_terminalCollectors.ConnectToAggregator("OutputString",
140 m_aggregator,
141 &MultiFileAggregator::AddContextHeading);
142
143 // Setup first-level collectors.
144 m_conversionCollectors.SetType("ns3::UnitConversionCollector");
145 m_conversionCollectors.SetAttribute("ConversionType",
146 EnumValue(UnitConversionCollector::FROM_BYTES_TO_KBIT));
147 CreateCollectorPerIdentifier(m_conversionCollectors);

152 }

154 case SatStatsHelper::OUTPUT_HISTOGRAM_FILE:
155 case SatStatsHelper::OUTPUT_PDF_FILE:
156 case SatStatsHelper::OUTPUT_CDF_FILE:
157 NS_FATAL_ERROR(GetOutputTypeName(GetOutputType())
158 << " is not a valid output type for this statistics.");
159 break;
160
161 case SatStatsHelper::OUTPUT_SCALAR_PLOT:
163 NS_FATAL_ERROR(GetOutputTypeName(GetOutputType())
164 << " is not a valid output type for this statistics.");
165 break;
166
167 case SatStatsHelper::OUTPUT_SCATTER_PLOT: {
168 // Setup aggregator.
169 m_aggregator = CreateAggregator("ns3::MagisterGnuplotAggregator",
170 "OutputPath",
171 StringValue(GetOutputPath()),
172 "OutputFileName",
173 StringValue(GetName()));
174 Ptr<MagisterGnuplotAggregator> plotAggregator =
175 m_aggregator->GetObject<MagisterGnuplotAggregator>();
176 NS_ASSERT(plotAggregator != nullptr);

179 plotAggregator->Set2dDatasetDefaultStyle(Gnuplot2dDataset::LINES);

182 m_terminalCollectors.SetType("ns3::IntervalRateCollector");
183 m_terminalCollectors.SetAttribute("InputDataType",
184 EnumValue(IntervalRateCollector::INPUT_DATA_TYPE_DOUBLE));
185 CreateCollectorPerIdentifier(m_terminalCollectors);

234void
235SatStatsSignallingLoadHelper::SignallingTxCallback(Ptr<const Packet> packet, const Address& to)
236{
237 // NS_LOG_FUNCTION (this << packet->GetSize () << from);
238
239 if (to.IsInvalid())
240 {
241 NS_LOG_WARN(this << " discarding packet " << packet << " (" << packet->GetSize()
242 << " bytes)" << " from statistics collection because of"
243 << " invalid sender address");
244 }
245 else
246 {
247 const Mac48Address addr = Mac48Address::ConvertFrom(to);
248
249 if (addr.IsBroadcast())
250 {
251 NS_LOG_INFO(this << " broadcast control message packet");
252
253 // Pass the sample to every first-level collectors.
254 for (CollectorMap::Iterator it = m_conversionCollectors.Begin();
255 it != m_conversionCollectors.End();
256 ++it)
257 {
258 Ptr<UnitConversionCollector> c = it->second->GetObject<UnitConversionCollector>();
259 NS_ASSERT(c != nullptr);
260 c->TraceSinkUinteger32(0, packet->GetSize());
261 }
262 }
263 else
264 {
265 // Determine the identifier associated with the sender address.
266 std::map<const Address, uint32_t>::const_iterator it = m_identifierMap.find(addr);
267
268 if (it == m_identifierMap.end())
269 {
270 NS_LOG_WARN(this << " discarding packet " << packet << " (" << packet->GetSize()
271 << " bytes)" << " from statistics collection because of"
272 << " unknown sender address " << addr);
273 }
274 else
275 {
276 // Find the first-level collector with the right identifier.
277 Ptr<DataCollectionObject> collector = m_conversionCollectors.Get(it->second);
278 NS_ASSERT_MSG(collector != nullptr,
279 "Unable to find collector with identifier " << it->second);
280 Ptr<UnitConversionCollector> c = collector->GetObject<UnitConversionCollector>();
281 NS_ASSERT(c != nullptr);
282
283 // Pass the sample to the collector.
284 c->TraceSinkUinteger32(0, packet->GetSize());
285 }
286 }
287 }
288
289} // end of `void RxCallback (Ptr<const Packet>, const Address);`

314void
315SatStatsFwdSignallingLoadHelper::DoInstallProbes()
316{
317 NS_LOG_FUNCTION(this);
318
319 // Create a map of UT addresses and identifiers.
320 NodeContainer uts = Singleton<SatTopology>::Get()->GetUtNodes();
321 for (NodeContainer::Iterator it = uts.Begin(); it != uts.End(); ++it)
322 {
323 SaveAddressAndIdentifier(*it);
324 }
325
326 // Connect to trace sources at GW nodes.
327
328 NodeContainer gws = Singleton<SatTopology>::Get()->GetGwNodes();
329 Callback<void, Ptr<const Packet>, const Address&> callback =
330 MakeCallback(&SatStatsFwdSignallingLoadHelper::SignallingTxCallback, this);
331
332 for (NodeContainer::Iterator it = gws.Begin(); it != gws.End(); ++it)
333 {
334 NetDeviceContainer devs = GetGwSatNetDevice(*it);
335
336 for (NetDeviceContainer::Iterator itDev = devs.Begin(); itDev != devs.End(); ++itDev)
337 {
338 NS_ASSERT((*itDev)->GetObject<SatNetDevice>() != nullptr);
339
340 if ((*itDev)->TraceConnectWithoutContext("SignallingTx", callback))
341 {
342 NS_LOG_INFO(this << " successfully connected with node ID " << (*it)->GetId()
343 << " device #" << (*itDev)->GetIfIndex());
344 }
345 else
346 {
347 NS_FATAL_ERROR("Error connecting to SignallingTx trace source"
348 << " of SatNetDevice" << " at node ID " << (*it)->GetId()
349 << " device #" << (*itDev)->GetIfIndex());
350 }
351
352 } // end of `for (NetDeviceContainer::Iterator itDev = devs)`
353
354 } // end of `for (NodeContainer::Iterator it = gws)`
355
356} // end of `void DoInstallProbes ();`

381void
382SatStatsRtnSignallingLoadHelper::DoInstallProbes()
383{
384 NS_LOG_FUNCTION(this);
385 NodeContainer uts = Singleton<SatTopology>::Get()->GetUtNodes();
386
387 for (NodeContainer::Iterator it = uts.Begin(); it != uts.End(); ++it)
388 {
389 const int32_t utId = GetUtId(*it);
390 NS_ASSERT_MSG(utId > 0, "Node " << (*it)->GetId() << " is not a valid UT");
391 const uint32_t identifier = GetIdentifierForUt(*it);
392
393 // Create the probe.
394 std::ostringstream probeName;
395 probeName << utId;
396 Ptr<ApplicationPacketProbe> probe = CreateObject<ApplicationPacketProbe>();
397 probe->SetName(probeName.str());
398
399 Ptr<NetDevice> dev = GetUtSatNetDevice(*it);
400
401 // Connect the object to the probe.
402 if (probe->ConnectByObject("SignallingTx", dev))
403 {
404 // Connect the probe to the right collector.
405 if (m_conversionCollectors.ConnectWithProbe(
406 probe->GetObject<Probe>(),
407 "OutputBytes",
408 identifier,
409 &UnitConversionCollector::TraceSinkUinteger32))
410 {
411 NS_LOG_INFO(this << " created probe " << probeName.str()
412 << ", connected to collector " << identifier);
413 m_probes.insert(
414 std::make_pair(probe->GetObject<Probe>(), std::make_pair(*it, identifier)));
415 }
416 else
417 {
418 NS_LOG_WARN(this << " unable to connect probe " << probeName.str()
419 << " to collector " << identifier);
420 }
421
422 } // end of `if (probe->ConnectByObject ("SignallingTx", dev))`
423 else
424 {
425 NS_FATAL_ERROR("Error connecting to SignallingTx trace source of SatNetDevice"
426 << " at node ID " << (*it)->GetId() << " device #2");
427 }
428
429 } // end of `for (it = uts.Begin(); it != uts.End (); ++it)`
430
431} // end of `void DoInstallProbes ();`

433void
434SatStatsRtnSignallingLoadHelper::UpdateIdentifierOnProbes()
435{
436 NS_LOG_FUNCTION(this);
437
438 std::map<Ptr<Probe>, std::pair<Ptr<Node>, uint32_t>>::iterator it;
439
440 for (it = m_probes.begin(); it != m_probes.end(); it++)
441 {
442 Ptr<Probe> probe = it->first;
443 Ptr<Node> node = it->second.first;
444 uint32_t identifier = it->second.second;
445
446 if (!m_conversionCollectors.DisconnectWithProbe(
447 probe->GetObject<Probe>(),
448 "OutputBytes",
449 identifier,
450 &UnitConversionCollector::TraceSinkUinteger32))
451 {
452 NS_FATAL_ERROR("Error disconnecting trace file on handover");
453 }
454
455 identifier = GetIdentifierForUtUser(node);
456
457 if (!m_conversionCollectors.ConnectWithProbe(probe->GetObject<Probe>(),
458 "OutputBytes",
459 identifier,
460 &UnitConversionCollector::TraceSinkUinteger32))
461 {
462 NS_FATAL_ERROR("Error connecting trace file on handover");
463 }
464
465 it->second.second = identifier;
466 }
467} // end of `void UpdateIdentifierOnProbes ();`
